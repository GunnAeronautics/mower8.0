#include <Wire.h>
#include <Adafruit_Sensor.h>
#include <Adafruit_ADXL343.h>

#define SAMPLES 200
#define G_MS2   9.80665f   // 1g in m/s^2 (library outputs m/s^2)

Adafruit_ADXL343 accel1 = Adafruit_ADXL343(12345);

struct Vec3 {
  float x, y, z;
};

Vec3 meas[6];   // +X, -X, +Y, -Y, +Z, -Z
Vec3 offset;
Vec3 scale;

/* Read averaged acceleration using Unified Sensor API */
Vec3 readAverage() {
  float sx = 0, sy = 0, sz = 0;
  sensors_event_t e;

  for (int i = 0; i < SAMPLES; i++) {
    accel1.getEvent(&e);

    sx += e.acceleration.x;
    sy += e.acceleration.y;
    sz += e.acceleration.z;

    delay(5);
  }

  return {
    sx / SAMPLES,
    sy / SAMPLES,
    sz / SAMPLES
  };
}

void waitForEnter() {
  while (!Serial.available());
  while (Serial.available()) Serial.read();
}

void prompt(const char* msg) {
  Serial.println();
  Serial.println(msg);
  Serial.println("Press ENTER when ready...");
  waitForEnter();
}

void setup() {
  Serial.begin(115200);
  Wire.begin(5, 22);

  if (!accel1.begin()) {
    Serial.println("ADXL343 not detected!");
    while (1);
  }

  accel1.setRange(ADXL343_RANGE_2_G);

  Serial.println("=== ADXL343 3D Calibration (Unified Sensor) ===");

  prompt("Place sensor with +X pointing UP");
  meas[0] = readAverage();

  prompt("Place sensor with -X pointing UP");
  meas[1] = readAverage();

  prompt("Place sensor with +Y pointing UP");
  meas[2] = readAverage();

  prompt("Place sensor with -Y pointing UP");
  meas[3] = readAverage();

  prompt("Place sensor with +Z pointing UP");
  meas[4] = readAverage();

  prompt("Place sensor with -Z pointing UP");
  meas[5] = readAverage();

  /* Offsets (m/s^2) */
  offset.x = (meas[0].x + meas[1].x) / 2.0f;
  offset.y = (meas[2].y + meas[3].y) / 2.0f;
  offset.z = (meas[4].z + meas[5].z) / 2.0f;

  /* Scale factors (convert measured delta to ±1g) */
  scale.x = (2.0f * G_MS2) / (meas[0].x - meas[1].x);
  scale.y = (2.0f * G_MS2) / (meas[2].y - meas[3].y);
  scale.z = (2.0f * G_MS2) / (meas[4].z - meas[5].z);

  Serial.println("\n=== CALIBRATION RESULTS ===");

  Serial.println("\nOffsets (m/s^2):");
  Serial.print("X_OFFSET = "); Serial.println(offset.x, 6);
  Serial.print("Y_OFFSET = "); Serial.println(offset.y, 6);
  Serial.print("Z_OFFSET = "); Serial.println(offset.z, 6);

  Serial.println("\nScale factors:");
  Serial.print("X_SCALE = "); Serial.println(scale.x, 6);
  Serial.print("Y_SCALE = "); Serial.println(scale.y, 6);
  Serial.print("Z_SCALE = "); Serial.println(scale.z, 6);

  Serial.println("\nApply calibration as:");
  Serial.println("(raw - offset) * scale");
}

void loop() {
  // Calibration complete
}
